5. Kanban View Component
--------------------------------------------------------------------------------
5.1. In client/src/components/KanbanView.tsx:
     ```jsx
     import { DragDropContext, Droppable, Draggable } from 'react-beautiful-dnd';
     import React from 'react';

     const STAGES = ['New','Contacted','Qualified','Appointment Set','Completed'];

     function KanbanView({ leadsByStage, updateLead, openFilters, openAddLead }) {
       const handleDragEnd = (result) => {
         const { source, destination, draggableId } = result;
         if (!destination) return;
         if (source.droppableId===destination.droppableId && source.index===destination.index) return;
         updateLead(draggableId, destination.droppableId);
       };

       return (
         <div className="flex flex-col h-full">
           <div className="flex space-x-2 p-2">
             <button onClick={openFilters} className="btn">Filters</button>
             <button onClick={openAddLead} className="btn-primary">Add Lead</button>
           </div>
           <DragDropContext onDragEnd={handleDragEnd}>
             <div className="flex overflow-x-auto space-x-4 p-2">
               {STAGES.map(stage => (
                 <Droppable key={stage} droppableId={stage}>
                   {(provided) => (
                     <div
                       ref={provided.innerRef}
                       {...provided.droppableProps}
                       className="w-64 bg-gray-100 rounded p-2 flex-shrink-0"
                     >
                       <h3 className="font-bold mb-2">{stage} ({leadsByStage[stage].length})</h3>
                       {leadsByStage[stage].map((lead, index) => (
                         <Draggable key={lead.id} draggableId={lead.id} index={index}>
                           {(prov, snap) => (
                             <div
                               ref={prov.innerRef}
                               {...prov.draggableProps}
                               {...prov.dragHandleProps}
                               className={`bg-white p-4 mb-2 rounded shadow ${
                                 snap.isDragging ? 'bg-blue-50 shadow-lg' : ''
                               }`}
                             >
                               <div className="font-semibold">{lead.name}</div>
                               <div className="text-sm text-gray-600">{lead.contact}</div>
                               <div className="mt-2 space-x-1">
                                 {STAGES.map(s => (
                                   <button
                                     key={s}
                                     onClick={()=>updateLead(lead.id, s)}
                                     className={`px-2 py-1 text-xs rounded ${
                                       lead.stage===s ? 'bg-blue-600 text-white' : 'bg-gray-200'
                                     }`}
                                   >{s}</button>
                                 ))}
                               </div>
                             </div>
                           )}
                         </Draggable>
                       ))}
                       {provided.placeholder}
                     </div>
                   )}
                 </Droppable>
               ))}
             </div>
           </DragDropContext>
         </div>
       );
     }

     export default KanbanView;
     ```

5.2. Styling:
     ```css

     .btn-primary { @apply px-4 py-2 bg-blue-600 text-white rounded; }
     ```
